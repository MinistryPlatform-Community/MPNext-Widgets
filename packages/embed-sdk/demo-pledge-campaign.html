<!-- Use the widget -->
<!-- Attributes: -->
<!--   campaign-id="N" the Pledge_Campaign_ID to display (or pass ?id=N in the URL); -->
<!--   pledge-email-template="N" dp_Communications ID for the confirmation email; -->
<!--   suggested-amounts="30,50,100" up to 3 quick-pick buttons ("NULL" to hide). -->
<next-pledge-campaign campaign-id="3" suggested-amounts="30,50,100"></next-pledge-campaign>

<!-- Listen to events -->
<script>
  const widget = document.querySelector("next-pledge-campaign");
  widget.addEventListener("pledgeCampaignLoaded", (e) => console.log("Loaded:", e.detail));
  widget.addEventListener("pledgeSaved", (e) => console.log("Saved:", e.detail.pledgeId));
  widget.addEventListener("pledgeError", (e) => console.error(e.detail));
</script>
